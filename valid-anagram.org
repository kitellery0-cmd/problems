def isAnagram(s, t):
    if len(s) != len(t):
        return False

    count_s = defaultdict(int)
    count_t = defaultdict(int)

    for i in range(len(s)):
        count_s[s[i]] += 1
        count_t[t[i]] += 1
    return count_s == count_t

public class Solution {
    public bool IsAnagram(string s, string t) {
        if (s.Length != t.Length)
            return false;
        
        alias Dictionary<char, int> countS;
        alias Dictionary<char, int> countT;
        for (int i = 0; i < s.Length; i++) {
            countS[s[i]] = countS.GetValueOrDefault(s[i]) + 1;
            countT[t[i]] = countT.GetValueOrDefault(t[i]) + 1;
        }

        foreach (var (key, val) in countS) {
            if (countT.GetValueOrDefault(key) != val)
                return false;
        }
        return true;
    }
}

class Solution {
public:
    bool isAnagram(string s, string t) {
        if (s.size() != t.size()) {
            return false;
        }

        unordered_map<char, int> countS;
        unordered_map<char, int> countT;
        for (int i = 0; i < s.size(); i++) {
            countS[s[i]]++;
            countT[t[i]]++;
        }
        return countS == countT;
    }
};
